Method Comparison EP09

obj <- mcr(d, "id", "candidate", "reference")Create a paired candidate-versus-reference object

obj <- correlation(obj, "pearson")Calculate Pearson, Spearman, or Kendall correlation
obj <- regression(obj, method = "deming", lambda = 1)Fit OLS, WLS, Deming, or Passing–Bablok regression
obj <- bland_altman(obj, type = "difference")Estimate the mean difference and limits of agreement
obj <- outlier(obj, type = "difference")Identify candidate outliers in paired differences
obj <- bias(obj, mdl = c(15, 25), interval = "both")Estimate bias and intervals at medical decision levels
predict(obj, reference = c(15, 25))Predict candidate results from reference values
plot(obj, type = "regression", interval = "confidence")Plot scatter, regression, Bland–Altman, or bias views
summary(obj)Summarize completed analysis steps and results

Precision EP05 / EP15

p <- precision(d, result ~ day/run, by = "sample")Create a repeatability and within-laboratory precision object
p <- outlier(p, method = "grubbs")Flag candidate outliers within sample or concentration groups
p <- normal(p, method = "auto")Assess normality within sample groups
p <- variance(p, method = "auto")Estimate variance components with ANOVA or REML
vc(p)Extract variance, SD, CV, and component shares
p <- ci(p)Calculate confidence intervals for SD and CV
p <- profile(p, model.no = 1:10)Fit a Sadler precision profile across concentration
report(p, day = "day", run = "run", table = "CI")Build a standardized precision report by day, run, and sample
plot(p, type = "var")Plot dot, histogram, variance, Q–Q, or profile views
list_sadler()List available Sadler profile models
